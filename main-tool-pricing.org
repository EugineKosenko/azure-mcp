#+title: Інструмент ~pricing~

Ціни зберігання ~Standard HDD Managed Disks~ у заданому регіоні: снепшоти (~LRS~ / ~ZRS~, за фактично
використаний ГБ на місяць) і рівні дисків (~S4~–~S80~, за виділений розмір, округлення вгору) — з
публічного =Azure Retail Prices= (без автентифікації, не =arm=, див. [[file:main-client.org][~main-client.org~]]). З необов'язковим
~size~ (напр. ~Standard_D4als_v6~, порівняй з [[file:main-tool-skus.org][~skus~]]) до відповіді додається ще й погодинна ціна цього
типу VM — окремо ~Linux~ і ~Windows~ (Windows дорожча через ліцензію). Аргумент ~region~
обов'язковий. Опис для ~tools/list~ і диспетчеризація — в [[file:main-tools.org][~main-tools.org~]].

* Рівні дисків

Порядок і розмір рівнів ~Standard HDD~ (~S4~–~S80~) =api= не повертає — це усталений, стабільний перелік
=Azure= (~Standard HDD size~, [[https://learn.microsoft.com/azure/virtual-machines/disks-types][disks-types]]).

* Таблиця

~meterName~ розрізняє три групи метрик того самого продукту: снепшоти (закінчується на ~ Snapshots~),
рівні диска (закінчується на ~LRS Disk~) і решта (операції вводу-виводу, конфіденційне шифрування) — її
не показуємо. Ціну образу (~Microsoft.Compute/images~) =Azure= окремо не документує: він міг би
тарифікуватися як диск за рівнем, міг би — як снепшот за використаними даними; про це прямо в тексті
відповіді, а не мовчки обираємо одну модель.

#+begin_src rust :noweb-ref Декларації
  use crate::client;
  use crate::tools;
#+end_src

#+begin_src rust :noweb-ref Визначення
  fn currency(items: &[serde_json::Value]) -> &str {
      items.first().map(|item| tools::text(item, "/currencyCode")).unwrap_or("-")
  }

  fn snapshots(items: &[serde_json::Value]) -> String {
      items
          .iter()
          .filter(|item| tools::text(item, "/meterName").ends_with(" Snapshots"))
          .map(|item| {
              format!(
                  "{:>8.3}  {}",
                  item["retailPrice"].as_f64().unwrap(),
                  tools::text(item, "/meterName").trim_end_matches(" Snapshots"),
              )
          })
          .collect::<Vec<_>>()
          .join("\n")
  }

  fn tiers(items: &[serde_json::Value]) -> String {
      TIERS
          .iter()
          .filter_map(|(tier, gib)| {
              items
                  .iter()
                  .find(|item| tools::text(item, "/meterName") == format!("{} LRS Disk", tier))
                  .map(|item| format!("{:>8.3}  {}  {} ГБ", item["retailPrice"].as_f64().unwrap(), tier, gib))
          })
          .collect::<Vec<_>>()
          .join("\n")
  }

  fn table(items: &[serde_json::Value]) -> String {
      if items.is_empty() {
          return "Цін не знайдено (перевірте код регіону).".to_string();
      }

      let currency = currency(items);

      format!(
          "Снепшоти (за фактично використаний ГБ, {currency} на місяць):\n{}\n\n\
           Рівні дисків (за виділений розмір, округлення вгору, {currency} на місяць):\n{}\n\n\
           Образ (Microsoft.Compute/images) Azure окремо не документує: можливо, тарифікується як диск за \
           рівнем (друга таблиця), можливо — як снепшот за використаними даними (перша); точну модель \
           перевірте на першому реальному образі інструментом costs.",
          snapshots(items), tiers(items),
      )
  }
#+end_src

* VM compute

Погодинна ціна конкретного типу VM — окрема пара запитів ~$filter~ (за ~armSkuName~), а не частина
основного: без ~size~ вона зайва. ~meterName~ без суфіксів ~ Spot~ / ~ Low Priority~ — звичайна (on-demand)
ціна; ~productName~ з ~Windows~ і без — дві різні ціни того самого розміру, перевірено живим викликом
2026-09-25 (~Standard_D4als_v6~ в ~eastus~: ~0.161~ =Linux=, ~0.345~ =Windows=).

#+begin_src rust :noweb-ref Визначення
  fn hourly(items: &[serde_json::Value], windows: bool) -> Option<f64> {
      items
          .iter()
          .filter(|item| !tools::text(item, "/meterName").contains("Spot"))
          .filter(|item| !tools::text(item, "/meterName").contains("Low Priority"))
          .find(|item| tools::text(item, "/productName").contains("Windows") == windows)
          .and_then(|item| item["retailPrice"].as_f64())
  }

  fn vm(items: &[serde_json::Value], size: &str) -> String {
      if items.is_empty() {
          return format!("VM {}: цін не знайдено (перевірте назву типу, напр. через skus).", size);
      }

      format!(
          "VM {} ({} на годину):\nLinux    {}\nWindows  {}",
          size,
          currency(items),
          hourly(items, false).map_or("-".to_string(), |price| format!("{:.3}", price)),
          hourly(items, true).map_or("-".to_string(), |price| format!("{:.3}", price)),
      )
  }
#+end_src

* Перевірки

Зразок — реальна відповідь =api= для ~eastus~ (перевірено живим запитом 2026-09-25), скорочена до кількох
метрик, разом з рядком ~Operations~, який має бути відкинутий.

          serde_json::json!({ "currencyCode": "USD", "retailPrice": 0.0005, "meterName": "S4 LRS Disk Operations" }),

  #[test]
  fn table_separates_snapshots_and_tiers() {
      let text = table(&items());

      assert!(text.contains("0.050  LRS"));
      assert!(text.contains("0.050  ZRS"));
      assert!(text.contains("1.536  S4  32 ГБ"));
      assert!(text.contains("3.008  S6  64 ГБ"));
      assert!(!text.contains("Operations"));
  }

  #[test]
  fn table_empty_is_a_hint() {
      assert_eq!(table(&[]), "Цін не знайдено (перевірте код регіону).".to_string());
  }
