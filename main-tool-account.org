#+title: Інструмент ~account~

Дані підписки: назва, ідентифікатор, стан, тенант, тип угоди (~quotaId~) і ліміт витрат
(~spendingLimit~) — заміна ~az account show~ прямим ~GET~ (~Subscriptions - Get~,
~api-version=2022-12-01~, перевірено живим викликом 2026-09-25). Один об'єкт, не перелік — див.
[[file:main-client.org][~client::get~]]. Опис для ~tools/list~ і диспетчеризація — в [[file:main-tools.org][~main-tools.org~]].

* Таблиця

Шість полів рядками ~ключ=значення~, а не одним рядком: тут це не перелік ресурсів, порівнювати з
рештою інструментів нема потреби.

#+begin_src rust :noweb-ref Декларації
  use crate::client;
  use crate::tools;
#+end_src

#+begin_src rust :noweb-ref Визначення
  fn table(body: &serde_json::Value) -> String {
      format!(
          "displayName={}\nsubscriptionId={}\nstate={}\ntenantId={}\nquotaId={}\nspendingLimit={}",
          tools::text(body, "/displayName"),
          tools::text(body, "/subscriptionId"),
          tools::text(body, "/state"),
          tools::text(body, "/tenantId"),
          tools::text(body, "/subscriptionPolicies/quotaId"),
          tools::text(body, "/subscriptionPolicies/spendingLimit"),
      )
  }
#+end_src

  let subscription = match tools::subscription(arguments).await {

  #[test]
  fn table_shows_fields() {
      let body = serde_json::json!({
          "displayName": "Demo Subscription",
          "subscriptionId": "11111111-1111-1111-1111-111111111111",
          "state": "Enabled",
          "tenantId": "22222222-2222-2222-2222-222222222222",
          "subscriptionPolicies": { "quotaId": "PayAsYouGo_2014-09-01", "spendingLimit": "Off" },
      });

      assert_eq!(
          table(&body),
          "displayName=Demo Subscription\n\
           subscriptionId=11111111-1111-1111-1111-111111111111\n\
           state=Enabled\n\
           tenantId=22222222-2222-2222-2222-222222222222\n\
           quotaId=PayAsYouGo_2014-09-01\n\
           spendingLimit=Off"
      );
  }
