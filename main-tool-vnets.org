#+title: Інструмент ~vnets~

Повертає віртуальні мережі й підмережі: адресний простір мережі, ім'я, адресний префікс і прив'язана NSG
кожної підмережі. Виклик — ~GET .../providers/Microsoft.Network/virtualNetworks~
(~api-version=2024-05-01~, перевірено живим викликом 2026-09-25). Заміна ~az network
vnet list~ разом із ~az network vnet subnet show ... networkSecurityGroup.id~: підмережа вже несе цей
ідентифікатор у власній відповіді, окремий запит на кожну не потрібен. Опис для ~tools/list~ і
диспетчеризація — в [[file:main-tools.org][~main-tools.org~]].

* Таблиця

Рядок — на кожну підмережу, не на кожну мережу: мереж зазвичай мало, підмереж — кілька, і саме
підмережа несе цікаві поля (префікс, NSG). Мережа без жодної підмережі (рідкість) все одно дає один
рядок, з ~subnet=-~. Адресний префікс підмережі =Azure= повертає то в однині
(~addressPrefix~), то в множині (~addressPrefixes~, коли задано кілька) — перевірено живим виводом
(в одній і тій самій мережі одна підмережа так, інша інак); ~prefix~ бере перше наявне. Підмережа без NSG —
~nsg=-~, це нормальний стан ([[file:main-tool-nsgrules.org][~nsgrules~]] показує прив'язку з боку самої NSG, цей інструмент — з боку
мережі). Фільтр ~name~ (ім'я мережі) не залежить від регістру.

#+begin_src rust :noweb-ref Визначення
  fn joined(value: &serde_json::Value, pointer: &str) -> String {
      let items: Vec<&str> = value.pointer(pointer).and_then(|item| item.as_array()).into_iter().flatten().filter_map(|item| item.as_str()).collect();

      if items.is_empty() { "-".to_string() } else { items.join(",") }
  }

  fn prefix(subnet: &serde_json::Value) -> String {
      match tools::text(subnet, "/properties/addressPrefix") {
          "-" => joined(subnet, "/properties/addressPrefixes"),
          text => text.to_string(),
      }
  }

  fn line(vnet: &serde_json::Value, subnet: Option<&serde_json::Value>) -> String {
      format!(
          "{}  {}  {}  subnet={}  prefix={}  nsg={}",
          tools::text(vnet, "/name"),
          tools::part(tools::text(vnet, "/id"), "resourceGroups").to_lowercase(),
          joined(vnet, "/properties/addressSpace/addressPrefixes"),
          subnet.map_or("-".to_string(), |subnet| tools::text(subnet, "/name").to_string()),
          subnet.map_or("-".to_string(), prefix),
          subnet.map_or("-".to_string(), |subnet| tools::last(tools::text(subnet, "/properties/networkSecurityGroup/id")).to_string()),
      )
  }

  fn table(vnets: &[serde_json::Value], name: Option<&str>) -> String {
      let lines: Vec<String> = tools::sorted(vnets)
          .into_iter()
          .filter(|vnet| name.is_none_or(|name| tools::same(tools::text(vnet, "/name"), name)))
          .flat_map(|vnet| {
              let mut subnets: Vec<&serde_json::Value> = vnet["properties"]["subnets"].as_array().into_iter().flatten().collect();
              subnets.sort_by_key(|subnet| tools::text(subnet, "/name").to_lowercase());

              if subnets.is_empty() {
                  vec![line(vnet, None)]
              } else {
                  subnets.into_iter().map(|subnet| line(vnet, Some(subnet))).collect()
              }
          })
          .collect();

      if lines.is_empty() { "Мереж не знайдено.".to_string() } else { lines.join("\n") }
  }
#+end_src

      Ok(scope) => scope,
      Err(message) => return tools::reply(Err(message)),

* Перевірки

Зразок — форма реальної відповіді =api= (перевірено живим викликом 2026-09-25, дані замінено на
вигадані): одна підмережа без NSG, одна з NSG, одна з ~addressPrefixes~ (множина) замість
~addressPrefix~.

  fn vnets() -> Vec<serde_json::Value> {

          "properties": {
              "addressSpace": { "addressPrefixes": ["10.0.0.0/16"] },

  #[test]
  fn table_shows_subnets_and_nsg() {
      assert_eq!(
          table(&vnets(), None),
          "demo-vnet  demo-rg  10.0.0.0/16  subnet=default  prefix=10.0.0.0/24  nsg=-\n\
           demo-vnet  demo-rg  10.0.0.0/16  subnet=snet-1  prefix=10.0.1.0/24  nsg=demo-nsg"
      );
  }

  #[test]
  fn table_filters_by_name() {
      assert!(table(&vnets(), Some("DEMO-VNET")).starts_with("demo-vnet"));
      assert_eq!(table(&vnets(), Some("other")), "Мереж не знайдено.".to_string());
  }
